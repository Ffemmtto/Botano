Burger Menu (Mobile)

HTML-структура для мобильного меню (логика в `index.html` или `documentation.js`).

HTML (Пример):

<!-- Кнопка бургер (появляется на < 680px) -->
<button class="nav-toggle" aria-label="Открыть меню">
  <svg>...</svg>
</button>

<!-- Навигация (скрыта по умолчанию) -->
<nav class="nav" id="main-nav">
  ...
</nav>

JavaScript (Пример):

// JS для переключения класса 'nav-open' у body или nav
const toggleBtn = document.querySelector('.nav-toggle');
const nav = document.querySelector('#main-nav');

toggleBtn.addEventListener('click', () => {
  nav.classList.toggle('nav-open');
});

Tabs (Вкладки)

Переключение контента по вкладкам. Требует JS.

Содержимое Tab 1...

Содержимое Tab 2...

HTML:

<div class="tabs-container">
  <div class="tab-buttons">
    <button class="tab-link active" onclick="openTab(event, 'tab1-main')">Вкладка 1</button>
    <button class="tab-link" onclick="openTab(event, 'tab2-main')">Вкладка 2</button>
  </div>
  <div id="tab1-main" class="tab-content active">
    <p>Содержимое для вкладки 1...</p>
  </div>
  <div id="tab2-main" class="tab-content">
    <p>Содержимое для вкладки 2...</p>
  </div>
</div>

CSS (Добавьте в `documentation.css`):

.tab-buttons { border-bottom: 1px solid #333; }
.tab-link {
  background: none;
  border: none;
  color: #888;
  padding: 10px 15px;
  cursor: pointer;
}
.tab-link.active {
  color: var(--accent1, #6f4cff);
  border-bottom: 2px solid var(--accent1, #6f4cff);
}
.tab-content {
  display: none;
  padding: 1rem;
  color: #fff;
}
.tab-content.active { display: block; }

JavaScript (Добавьте в `documentation.js`):

function openTab(evt, tabId) {
  const container = evt.target.closest('.tabs-container, .tabs-preview');
  
  // Скрыть весь контент
  container.querySelectorAll('.tab-content').forEach(tab => {
    tab.style.display = 'none';
  });
  
  // Снять .active с кнопок
  container.querySelectorAll('.tab-link').forEach(link => {
    link.classList.remove('active');
  });
  
  // Показать нужный контент и активировать кнопку
  document.getElementById(tabId).style.display = 'block';
  evt.currentTarget.classList.add('active');
}

Modal (Модальное окно)

Система модальных окон. Требует JS.

HTML:

<!-- Кнопка-триггер -->
<button onclick="openModal('my-modal-id')">Открыть окно</button>

<!-- Фон (один на всю страницу) -->
<div class="modal-bg" id="modal-bg" onclick="closeModal(event)">

  <!-- Само модальное окно -->
  <div class="modal" id="my-modal-id-modal">
    <h2>Заголовок</h2>
    <p>Контент...</p>
    <button onclick="closeModal(event, true)">Закрыть</button>
  </div>

</div>

JavaScript (Добавьте в `documentation.js`):

function openModal(id) {
  document.getElementById('modal-bg').style.display = 'flex';
  document.getElementById(id + '-modal').style.display = 'block';
}

function closeModal(e, force = false) {
  // force = true для кнопки "Закрыть"
  if (e.target.id === 'modal-bg' || force) {
    document.getElementById('modal-bg').style.display = 'none';
    // Скрыть ВСЕ модалки
    document.querySelectorAll('.modal').forEach(m => {
      m.style.display = 'none';
    });
  }
  // Предотвращаем закрытие при клике на контент
  e.stopPropagation();
}
